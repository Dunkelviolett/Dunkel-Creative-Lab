    <script>
        // 1. Control de pestañas (Page / Discussion)
        function switchTab(tab) {
            const pageBtn = document.getElementById('tabPageBtn');
            const discBtn = document.getElementById('tabDiscBtn');
            const pageContent = document.getElementById('tabPageContent');
            const discContent = document.getElementById('tabDiscContent');

            if (tab === 'page') {
                pageBtn.className = "px-4 py-2 font-medium border-b-2 border-wikiGold text-white bg-wikiDark transition-colors";
                discBtn.className = "px-4 py-2 font-medium border-b-2 border-transparent text-zinc-400 hover:text-white transition-colors";
                pageContent.classList.remove('hidden');
                discContent.classList.add('hidden');
            } else {
                discBtn.className = "px-4 py-2 font-medium border-b-2 border-wikiGold text-white bg-wikiDark transition-colors";
                pageBtn.className = "px-4 py-2 font-medium border-b-2 border-transparent text-zinc-400 hover:text-white transition-colors";
                discContent.classList.remove('hidden');
                pageContent.classList.add('hidden');
            }
        }

        // 2. Toggle de la barra lateral en dispositivos móviles
        const sidebarToggle = document.getElementById('sidebarToggle');
        const sidebar = document.getElementById('sidebar');

        sidebarToggle.addEventListener('click', () => {
            sidebar.classList.toggle('hidden');
        });

        // 3. Sistema de cambio de Tema Oscuro / Claro
        const themeToggle = document.getElementById('themeToggle');
        const themeIcon = document.getElementById('themeIcon');
        const htmlElement = document.documentElement;

        themeToggle.addEventListener('click', () => {
            if (htmlElement.classList.contains('dark')) {
                htmlElement.classList.remove('dark');
                themeIcon.classList.replace('fa-moon', 'fa-sun');
                // Ajustes sencillos para modo claro dinámico
                document.body.style.backgroundColor = '#f4f4f5';
                document.body.style.color = '#18181b';
            } else {
                htmlElement.classList.add('dark');
                themeIcon.classList.replace('fa-sun', 'fa-moon');
                document.body.style.backgroundColor = '#121212';
                document.body.style.color = '#e4e4e7';
            }
        });
    </script>
